cmake_minimum_required(VERSION 3.25)
project(imvim VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(IMVIM_GUI "Build the graphical frontend" ON)
find_package(msgpack-cxx CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(imvim_core src/process.cpp src/session.cpp src/grid.cpp)
target_include_directories(imvim_core PUBLIC src)
target_link_libraries(imvim_core PUBLIC msgpack-cxx Threads::Threads)
if(WIN32)
  target_compile_definitions(imvim_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
endif()
if(MSVC)
  target_compile_options(imvim_core PUBLIC /W4 /utf-8)
else()
  target_compile_options(imvim_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
if(IMVIM_GUI)
  find_package(imgui CONFIG REQUIRED)
  add_executable(imvim src/main.cpp src/editor.cpp)
  target_link_libraries(imvim PRIVATE imvim_core imgui::imgui)
  if(WIN32)
    target_sources(imvim PRIVATE src/platform_win32.cpp)
    target_link_libraries(imvim PRIVATE d3d11 dxgi imm32 shell32)
    # Keep main() for Unicode argv conversion, but do not allocate a console.
    if(MSVC)
      target_link_options(imvim PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
    endif()
  else()
    find_package(SDL3 CONFIG REQUIRED)
    find_package(OpenGL REQUIRED)
    target_sources(imvim PRIVATE src/platform_sdl.cpp)
    target_link_libraries(imvim PRIVATE SDL3::SDL3 OpenGL::GL)
  endif()
  add_custom_command(TARGET imvim POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
      "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts" "$<TARGET_FILE_DIR:imvim>/fonts")
  install(TARGETS imvim RUNTIME DESTINATION .)
  install(DIRECTORY assets/fonts DESTINATION .)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(imvim_tests tests/core_tests.cpp)
  target_link_libraries(imvim_tests PRIVATE imvim_core)
  add_test(NAME core COMMAND imvim_tests)
  add_test(NAME nvim_roundtrip COMMAND imvim_tests --integration)
  set_tests_properties(nvim_roundtrip PROPERTIES TIMEOUT 20)
endif()
